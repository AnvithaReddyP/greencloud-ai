        // Core Calculation Engine
        function calculateCalculatedMetrics() {
            const gridIntensity = REGIONAL_GRID[state.region] || 380;
            gridFactorVal.textContent = `${gridIntensity} gCO2 / kWh`;

            const estimatedWatts = (state.vcpu * 10) + (state.ram * 1.5) + (state.storage * 0.05);
            const monthlyKwh = Math.round((estimatedWatts * 730) / 1000);
            estKwEnergy.textContent = `${monthlyKwh} kWh /mo`;

            let evaluatedProviders = CLOUD_PROVIDERS.map(p => {
                const rawCost = (state.vcpu * p.cpuRate) + (state.ram * p.ramRate) + (state.storage * p.storageRate);
                const monthlyCost = parseFloat(rawCost.toFixed(2));

                const totalEnergyKwh = monthlyKwh * p.pue;
                const nonRenewableFraction = 1 - (p.renewablePct / 100);
                const carbonKg = parseFloat(((totalEnergyKwh * gridIntensity * nonRenewableFraction) / 1000).toFixed(1));

                let costFactor = Math.max(0, 100 - ((monthlyCost / state.budget) * 40));
                if (monthlyCost <= state.budget) costFactor += 20;

                let carbonFactor = Math.max(0, 100 - (carbonKg * 2));
                let renewableFactor = p.renewablePct;

                let matchScore = 0;
                if (state.priority === 'cost') {
                    matchScore = (costFactor * 0.6) + (carbonFactor * 0.2) + (renewableFactor * 0.2);
                } else if (state.priority === 'carbon') {
                    matchScore = (carbonFactor * 0.5) + (renewableFactor * 0.3) + (costFactor * 0.2);
                } else { // balanced
                    matchScore = (costFactor * 0.4) + (carbonFactor * 0.3) + (renewableFactor * 0.3);
                }

                matchScore = Math.min(99, Math.max(40, Math.round(matchScore)));

                return {
                    ...p,
                    monthlyCost,
                    carbonKg,
                    matchScore,
                    isOverBudget: monthlyCost > state.budget
                };
            });

            evaluatedProviders.sort((a, b) => b.matchScore - a.matchScore);

            const topMatch = evaluatedProviders[0];
            topProviderName.textContent = topMatch.name;
            topMatchBadge.textContent = `${topMatch.matchScore}% Match`;
            topProviderCost.textContent = `$${topMatch.monthlyCost.toFixed(2)}`;
            topProviderCarbon.textContent = `${topMatch.carbonKg} kg CO2`;
            topProviderPue.textContent = topMatch.pue.toFixed(2);
            topProviderReason.textContent = `${topMatch.name} delivers optimal balance for ${state.vcpu} vCPUs & ${state.ram}GB RAM in ${getRegionName(state.region)}.`;

            const cheapest = [...evaluatedProviders].sort((a, b) => a.monthlyCost - b.monthlyCost)[0];
            const greenest = [...evaluatedProviders].sort((a, b) => a.carbonKg - b.carbonKg)[0];

            cheapestName.textContent = cheapest.name;
            cheapestCost.textContent = `$${cheapest.monthlyCost.toFixed(2)} / mo`;
            greenestName.textContent = greenest.name;
            greenestCarbon.textContent = `${greenest.carbonKg} kg CO2 / mo`;

            renderProvidersGrid(evaluatedProviders);
            updateAgentContext(topMatch);
        }

        async function handleChatSubmit(promptText) {
            const userText = promptText || chatInput.value.trim();
            if (!userText) return;

            appendMessage('user', userText);
            chatInput.value = '';

            const typingId = 'typing-' + Date.now();
            appendTypingIndicator(typingId);

            const workloadContext = {
                vcpu: state.vcpu,
                ram: state.ram,
                storage: state.storage,
                budget: state.budget,
                region: getRegionName(state.region),
                cheapest: cheapestName.textContent,
                greenest: greenestName.textContent
            };

            try {
                const res = await fetch(`${API_BASE_URL}/api/chat`, {
                    method: 'POST',
                    headers: { 'Content-Type': 'application/json' },
                    body: JSON.stringify({
                        prompt: userText,
                        context: workloadContext
                    })
                });

                removeTypingIndicator(typingId);

                if (res.ok) {
                    const data = await res.json();
                    appendMessage('ai', data.response);
                } else {
                    const errData = await res.json().catch(() => ({ detail: 'Service offline' }));
                    appendMessage('ai', `I was unable to process your query via the backend. (${errData.detail || 'Offline mode'}). Please verify Uvicorn is running at http://127.0.0.1:8000.`);
                }
            } catch (err) {
                removeTypingIndicator(typingId);
                appendMessage('ai', "I was unable to retrieve a detailed recommendation from the backend server. Please make sure the Python server is running on `127.0.0.1:8000` or try again.");
            }
        }
